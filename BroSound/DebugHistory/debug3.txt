class Text
{
    List<Letter> letters = new List<Letter>();
    public Text(string path)
    {
        foreach (var file in Directory.GetFiles(path))
        {
            letters.Add(new Letter(file.Remove(0, file.LastIndexOf("\\") + 1).Replace(".png", ""), Raylib.LoadTexture(file)));
        }

    }
    public void WriteText(string text, float scale, Raylib_cs.Color color, System.Numerics.Vector2 Pos,out float EndLetterPos)
    {
        EndLetterPos = 0;
        int XMove = (int)((-text.Length * 0.5f) * 43 * scale);
        text = text.ToUpper();
        if((Pos.X + XMove) < 0)
        {
            XMove = (int)-Pos.X;
        }
        for(int x = 0; x < text.Length; x++)
        {
            string letter = text[x].ToString();
            if (text[x] == ' ')
            {
                letter = "Space";
            }
        read:;
            var texture = letters.Where(a => a.Name == letter).FirstOrDefault();
            if (texture != null)
            {
                Raylib.DrawTextureEx(texture.Texture, new System.Numerics.Vector2(Pos.X + XMove, Pos.Y), 0, scale, color);
                XMove += (int)(43 * scale);
                EndLetterPos = XMove + Pos.X;
            }
            else
            {
                letter = "Space";
                goto read;
            }
        }
       
        
    }
}

class TimeLine
{  
    public void Draw(Texture2D BoxPiece, float Value, float MaxValue, float Width, Vector2 Pos, Action clickAction)
    {
        BoxPiece.Width = (int)(Width / MaxValue);
        float BoxWidth = BoxPiece.Width;
        int Boxes = 0;
        int limit = 100000;
        for(int x = 0; x < Width; x += (int)BoxWidth)
        {
            Boxes++;
            limit--;
            if(limit == 0)
            {
                break;
            }
        }
        Boxes = Math.Clamp(Boxes, 0, (int)MaxValue);
        float move = -((Boxes * BoxWidth) / 2);
        for (int x = 0; x < Boxes; x++)
        {
            Button button = new Button(Environment.CurrentDirectory + "\\Fonts", true);
            if (Value > x)
            {
                //Raylib.DrawTextureEx(BoxPiece, new Vector2(Pos.X + move, Pos.Y), 0, 1, Color.Red);
                Action connected = clickAction += () => { Raylib.SeekMusicStream(Program.currentSound, x); };
                button.Interact(BoxPiece, new Vector2(Pos.X + move, Pos.Y), 1f, 1f, Raylib_cs.Color.Red, Raylib_cs.Color.White, "", BoxPiece.Height, BoxPiece.Width, connected, Raylib_cs.MouseButton.Left, 0, 0, out Program.MixOut);
            }
            else
            {
                //Raylib.DrawTextureEx(BoxPiece, new Vector2(Pos.X + move, Pos.Y), 0, 1, Color.White);
                Action connected = clickAction += () => { Raylib.SeekMusicStream(Program.currentSound, x); };
                button.Interact(BoxPiece, new Vector2(Pos.X + move, Pos.Y), 1f, 1f, Raylib_cs.Color.White, Raylib_cs.Color.White, "", BoxPiece.Height, BoxPiece.Width, connected, Raylib_cs.MouseButton.Left, 0, 0, out Program.MixOut);
            }
           
           
            move += BoxWidth;      
        }
    }
}
class ButtonList
{
    float virtualPosY = 1;
    public void Draw(List<Button> Buttons, float RaylibDeltaTime, int Height, System.Numerics.Vector2 Pos, float distance)
    {
        virtualPosY += Raylib.GetMouseWheelMove() * RaylibDeltaTime * 300;
        int cap = 0;
        int move = 0;
        foreach(var button in Buttons)
        {
            if((Math.Abs(Pos.Y - move + virtualPosY) > Pos.Y) && (Math.Abs(Pos.Y - move + virtualPosY) < Pos.Y + Height))
            {
                    //button.LastButtonTexture.Width = (int)(button.LastEndLetterPos - Pos.X);
                    //button.LastButtonColor = Raylib_cs.Color.Red;
                    button.Interact(button.LastButtonTexture, new System.Numerics.Vector2(Pos.X, Pos.Y - move + virtualPosY), button.LastButtonScale, button.LastTextScale, button.LastButtonColor, button.LastTextColor, button.LastText, button.LastButtonHeight, button.LastButtonWidth, button.LastAction, button.LastMouseButoon, button.LastLettersMoveY, button.LastButtonMoveY, out button.LastEndLetterPos);                              
            }
            else
            {
              
                                                        
            }
            
            move -= (int)(button.LastTextureHeight * distance);
            cap -= (int)(button.LastTextureHeight * distance);
        }
        if(virtualPosY < cap || virtualPosY > 50 * distance)
        {
            virtualPosY = 10;
        }
    }
}
