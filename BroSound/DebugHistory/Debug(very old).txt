using Raylib_cs;
using SharpHook;
using SharpHook.Data;
using System;
using System.Diagnostics;
using System.IO;
using System.Linq.Expressions;
using System.Numerics;
class Program
{
    static void Main(string[] args)
    {
        float MusicTimer = 0;
        bool isPause = false;
        Raylib.InitWindow(1000, 1000, "BroSound");
        Raylib.InitAudioDevice();
        Raylib.SetTargetFPS(60);
        Text text = new Text(Environment.CurrentDirectory + "\\Fonts");
        Button pausePlay = new Button(Environment.CurrentDirectory + "\\Fonts");
        Button nextMusic = new Button(Environment.CurrentDirectory + "\\Fonts");
        Button previousMusic = new Button(Environment.CurrentDirectory + "\\Fonts");
        TimeLine musicTime = new TimeLine();
        Texture2D MusicTimeBoxTexture = Raylib.LoadTexture(Environment.CurrentDirectory + "\\TimeLine" + "\\BasicBox.png");
        Texture2D UnviersalButtonTexture = Raylib.LoadTexture(Environment.CurrentDirectory + "\\Buttons" + "\\Basic.png");
        Texture2D PauseButtonTexture = Raylib.LoadTexture(Environment.CurrentDirectory + "\\Buttons" + "\\Pause.png");
        Texture2D PlayButtonTexture = Raylib.LoadTexture(Environment.CurrentDirectory + "\\Buttons" + "\\Play.png");
        Texture2D PreviousButtonTexture = Raylib.LoadTexture(Environment.CurrentDirectory + "\\Buttons" + "\\Previous.png");
        Texture2D NextButtonTexture = Raylib.LoadTexture(Environment.CurrentDirectory + "\\Buttons" + "\\Next.png");
        var Camera = new Camera2D() { Target = new System.Numerics.Vector2(0, 0), Rotation = 0 };
        string[] Files = Directory.GetFiles(Environment.CurrentDirectory + "\\Sounds");
        //Sound[] Sounds = new Sound[Files.Length];
        var hook = new SharpHook.TaskPoolGlobalHook();
        hook.RunAsync();
        int music = 0;
        //for (int i = 0; i <= music + 2; i++)
        //{
        //Sounds[i] = Raylib.LoadSound(Files[i]);
        //}
        
        Sound currentSound = Raylib.LoadSound(Files[music]);
        Raylib.PlaySound(currentSound);
        if (hook.IsRunning)
        {
            hook.KeyPressed += (sender, e) =>
            {
                if (e.Data.KeyCode == KeyCode.VcMediaNext)
                {   
                    Raylib.StopSound(currentSound);
                    Raylib.UnloadSound(currentSound);
                    music += 1;
                    if (music >= Files.Length)
                    {
                        music = 0;
                    }                   
                    currentSound = Raylib.LoadSound(Files[music]);
                    Raylib.PlaySound(currentSound);
                    Console.WriteLine("Next");
                    MusicTimer = 0;
                }
                if (e.Data.KeyCode == KeyCode.VcMediaPlay)
                {
                    if(Raylib.IsSoundPlaying(currentSound))
                    {
                        Raylib.PauseSound(currentSound);
                        Console.WriteLine("Pause");
                        isPause = true;
                    }
                    else
                    {
                        Raylib.ResumeSound(currentSound);
                        Console.WriteLine("Resume");
                        isPause = false;
                    }                   
                }
                if(e.Data.KeyCode == KeyCode.VcF11)
                {
                    Raylib.ToggleFullscreen();
                    Raylib.SetWindowSize(Raylib.GetMonitorWidth(Raylib.GetCurrentMonitor()), Raylib.GetMonitorHeight(Raylib.GetCurrentMonitor()));
                    if(Raylib.IsWindowFullscreen() == false)
                    {
                        Raylib.SetWindowSize(1000, 1000);
                    }
                }
            };
        }
      
        while (!Raylib.WindowShouldClose())
        {
            
            if(isPause == false)
            {
                MusicTimer += Raylib.GetFrameTime();
            }
                
            
           
            string musicTitle = Files[music].Remove(0, Files[music].LastIndexOf("\\") + 1);
            Raylib.BeginDrawing();

            Raylib.ClearBackground(Color.Gold);
            Raylib.BeginMode2D(Camera);
            Raylib.EndMode2D();

            text.WriteText(musicTitle, 1, Color.Blue, new System.Numerics.Vector2(Raylib.GetScreenWidth() * 0.5f, Raylib.GetScreenHeight() * 0.5f));
            if(Raylib.IsSoundPlaying(currentSound))
            {
                pausePlay.Interact(PauseButtonTexture, new System.Numerics.Vector2(Raylib.GetScreenWidth() * 0.5f, Raylib.GetScreenHeight() * 0.75f), 1f, 1f, Color.Green, Color.Red, "", 100, 100, ()=> {  Raylib.PauseSound(currentSound); isPause = true; }, Raylib_cs.MouseButton.Left);
            }
            else
            {
                pausePlay.Interact(PlayButtonTexture, new System.Numerics.Vector2(Raylib.GetScreenWidth() * 0.5f, Raylib.GetScreenHeight() * 0.75f), 1f, 1f, Color.Green, Color.Red, "", 100, 100, () => {  Raylib.ResumeSound(currentSound); isPause = false; }, Raylib_cs.MouseButton.Left);
            }
            previousMusic.Interact(PreviousButtonTexture, new System.Numerics.Vector2(Raylib.GetScreenWidth() * 0.25f, Raylib.GetScreenHeight() * 0.75f), 1f, 1f, Color.Green, Color.Red, "", 100, 100, () => { Raylib.StopSound(currentSound); Raylib.UnloadSound(currentSound); music--; if (music < 0) { music = Files.Length - 1; } currentSound = Raylib.LoadSound(Files[music]); Raylib.PlaySound(currentSound); }, Raylib_cs.MouseButton.Left);
            nextMusic.Interact(NextButtonTexture, new System.Numerics.Vector2(Raylib.GetScreenWidth() * 0.75f, Raylib.GetScreenHeight() * 0.75f), 1f, 1f, Color.Green, Color.Red, "", 100, 100, () => { Raylib.StopSound(currentSound); Raylib.UnloadSound(currentSound); music++; if (Files.Length < (music + 1)) { music = 0; } currentSound = Raylib.LoadSound(Files[music]); Raylib.PlaySound(currentSound); }, Raylib_cs.MouseButton.Left);
            try
            {
                MusicTimeBoxTexture.Width = 50;
                MusicTimeBoxTexture.Height = 50;
                musicTime.Draw(MusicTimeBoxTexture, MusicTimer, (float)currentSound.FrameCount / currentSound.Stream.SampleRate, 1000, new System.Numerics.Vector2(Raylib.GetScreenWidth() * 0.5f, Raylib.GetScreenHeight() * 0.6f), ()=> {  });
            }
            catch
            {

            }
            
            Raylib.EndDrawing();
        }
        Raylib.CloseAudioDevice();
        Raylib.CloseWindow();
    }
}
class Text
{
    List<Letter> letters = new List<Letter>();
    public Text(string path)
    {
        foreach (var file in Directory.GetFiles(path))
        {
            letters.Add(new Letter(file.Remove(0, file.LastIndexOf("\\") + 1).Replace(".png", ""), Raylib.LoadTexture(file)));
        }

    }
    public void WriteText(string text, float scale, Color color, System.Numerics.Vector2 Pos)
    {
        int XMove = (int)((-text.Length * 0.5f) * 43 * scale);
        text = text.ToUpper();
        if((Pos.X + XMove) < 0)
        {
            XMove = (int)-Pos.X;
        }
        for(int x = 0; x < text.Length; x++)
        {
            string letter = text[x].ToString();
            if (text[x] == ' ')
            {
                letter = "Space";
            }
        read:;
            var texture = letters.Where(a => a.Name == letter).FirstOrDefault();
            if (texture != null)
            {
                Raylib.DrawTextureEx(texture.Texture, new System.Numerics.Vector2(Pos.X + XMove, Pos.Y), 0, scale, color);
                XMove += (int)(43 * scale);
            }
            else
            {
                letter = "Space";
                goto read;
            }
        }
       
        
    }
}

    {
        Name = name;
        Texture = texture;
    }
    public string Name { get; set; }
    public Texture2D Texture { get; set; }
}
class Button
{
    Text Mtext;
    public Button(string LettersPath)
    {
        Mtext = new Text(LettersPath);
    }
   public void Interact(Texture2D buttonTexture, System.Numerics.Vector2 pos, float buttonScale, float textScale, Color buttonColor, Color textColor, string text, int buttonHeight, int buttonWidth, Action function, Raylib_cs.MouseButton interactButton)
   {
        buttonTexture.Height = buttonHeight;
        buttonTexture.Width = buttonWidth;
        Raylib.DrawTextureEx(buttonTexture, new System.Numerics.Vector2(pos.X - (buttonTexture.Width * 0.5f), pos.Y - (buttonTexture.Height * 0.5f)), 0, buttonScale, buttonColor);
        Mtext.WriteText(text, textScale, textColor, pos);
        Vector2 Center = new System.Numerics.Vector2(pos.X, pos.Y);
        Vector2 ButtonLeftUp = new Vector2(Center.X - (buttonTexture.Width * 0.5f), Center.Y - (buttonTexture.Height * 0.5f));
        Vector2 ButtonRightUp = new Vector2(Center.X + (buttonTexture.Width * 0.5f), Center.Y - (buttonTexture.Height * 0.5f));
        Vector2 ButtonLeftDown = new Vector2(Center.X - (buttonTexture.Width * 0.5f), Center.Y + (buttonTexture.Height * 0.5f));
        Vector2 ButtonRightDown = new Vector2(Center.X + (buttonTexture.Width * 0.5f), Center.Y + (buttonTexture.Height * 0.5f));
        if((ButtonLeftUp.Y < Raylib.GetMouseY()))
        {
            if ((ButtonRightUp.Y < Raylib.GetMouseY()))
            {
                if ((ButtonLeftDown.Y > Raylib.GetMouseY()))
                {
                    if ((ButtonRightDown.Y > Raylib.GetMouseY()))
                    {
                        if ((ButtonRightUp.X > Raylib.GetMouseX()))
                        {
                            if ((ButtonRightDown.X > Raylib.GetMouseX()))
                            {
                                if ((ButtonLeftUp.X < Raylib.GetMouseX()))
                                {
                                    if ((ButtonLeftDown.X < Raylib.GetMouseX()))
                                    {
                                        if(Raylib.IsMouseButtonPressed(interactButton))
                                        {
                                            function();
                                        }
                                        
                                    }
                                }
                            }
                        }




                    }
                }
            }
        }
    }
}
class TimeLine
{
    public void Draw(Texture2D BoxPiece, float Value, float MaxValue, float Width, Vector2 Pos, Action clickAction)
    {
        BoxPiece.Width = (int)(Width / MaxValue);
        float BoxWidth = BoxPiece.Width;
        int Boxes = 0;
        int limit = 10000;
        for(int x = 0; x < Width; x += (int)BoxWidth)
        {
            Boxes++;
            limit--;
            if(limit == 0)
            {
                break;
            }
        }
        float move = -((Boxes * BoxWidth) / 2);
        for (int x = 0; x < Boxes; x++)
        {
            Button button = new Button(Environment.CurrentDirectory = "\\Fonts");
            if (Value > x)
            {
                //Raylib.DrawTextureEx(BoxPiece, new Vector2(Pos.X + move, Pos.Y), 0, 1, Color.Red);
                button.Interact(BoxPiece, new Vector2(Pos.X + move, Pos.Y), 1f, 1f, Color.Red, Color.White, "", BoxPiece.Height, BoxPiece.Width, clickAction, Raylib_cs.MouseButton.Left);
            }
            else
            {
                Raylib.DrawTextureEx(BoxPiece, new Vector2(Pos.X + move, Pos.Y), 0, 1, Color.White);
            }
           
           
            move += BoxWidth;      
        }
    }
}
